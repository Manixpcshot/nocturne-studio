cmake_minimum_required(VERSION 3.22)
project(nocturne_engine LANGUAGES C CXX)

if(NOT EXISTS "${NOCTURNE_ENGINE}/ggml/CMakeLists.txt")
    message(FATAL_ERROR "Fetch the pinned stable-diffusion.cpp source and its ggml submodule before building.")
endif()

set(SD_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SD_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(SD_WEBP OFF CACHE BOOL "" FORCE)
set(SD_WEBM OFF CACHE BOOL "" FORCE)
option(NOCTURNE_VULKAN "Build the separate Vulkan engine" OFF)
set(SD_VULKAN ${NOCTURNE_VULKAN} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOCTURNE_VULKAN)
    find_package(SPIRV-Headers CONFIG REQUIRED)
endif()
add_subdirectory("${NOCTURNE_ENGINE}" engine)
add_library(nocturne_engine SHARED engine.cpp)
if(NOCTURNE_VULKAN)
    target_link_libraries(ggml-vulkan PRIVATE SPIRV-Headers::SPIRV-Headers)
    set_target_properties(nocturne_engine PROPERTIES OUTPUT_NAME nocturne_vulkan)
endif()
target_compile_features(nocturne_engine PRIVATE cxx_std_17)
target_link_libraries(nocturne_engine PRIVATE stable-diffusion log)
